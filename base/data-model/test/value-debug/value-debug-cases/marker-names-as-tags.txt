{
  // Each of the conversion's marker names is also a tag, which an
  // `UnknownValue` can carry over any payload. The marker takes precedence
  // where the payload has the marker's shape, and otherwise the value renders
  // as an instance.
  "circle over a depth": new UnknownValue("circle", 1),
  "circle over a string": new UnknownValue("circle", "x"),
  "uniqueSymbol over no description": new UnknownValue("uniqueSymbol", undefined),
  "uniqueSymbol over a description": new UnknownValue("uniqueSymbol", "d"),
  "uniqueSymbol over a record holding a bigint": new UnknownValue("uniqueSymbol", { x: 2n }),
  "partialString over a number": new UnknownValue("partialString", 5),
  "partialString over a record of another shape": new UnknownValue("partialString", { a: 1 }),
}

--

<circle>
<circle>

}

/circle("x")

{
  "/circle": "x"
}

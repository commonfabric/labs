(() => {
  // A real stack names this file and a line in it, so each error that has
  // one gets a representative one instead.
  const withStack = (error) => {
    error.stack = `${error.name}: ${error.message}\n` +
      "    at frob (file:///example/frob.ts:12:34)";
    return error;
  };
  const bare = (message) =>
    new FabricError({ type: "Error", message, stack: undefined, cause: undefined });
  return {
    plain: FabricError.fromNativeError(withStack(new Error("boom"))),
    bare: bare("boom"),
    "type other than `Error`": FabricError.fromNativeError(
      withStack(new TypeError("not a thing")),
    ),
    "name differs from type": new FabricError({
      type: "TypeError",
      name: "FrobError",
      message: "frob failed",
      stack: undefined,
      cause: undefined,
    }),
    "with native cause": FabricError.fromNativeError(
      withStack(new Error("outer", { cause: withStack(new Error("inner")) })),
    ),
    "with fabric cause": new FabricError({
      type: "Error",
      message: "outer",
      stack: undefined,
      cause: { detail: 1, where: ["x", 2] },
    }),
    "with extras": FabricError.fromNativeError(
      Object.assign(withStack(new Error("extra")), {
        code: 42,
        when: new Date("2026-09-03T12:34:56.789Z"),
      }),
    ),
    "cause chain": FabricError.fromNativeError(
      withStack(new Error("top", {
        cause: withStack(new Error("middle", {
          cause: withStack(new Error("root")),
        })),
      })),
    ),
  };
})()

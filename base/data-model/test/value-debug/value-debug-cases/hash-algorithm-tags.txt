{
  // A hash renders in one shape whatever its algorithm tag, including a tag
  // its codec refuses to encode for being outside the tag syntax.
  "tag fid1": new FabricHash(new Uint8Array([1, 2]), "fid1"),
  "tag outside the syntax": new FabricHash(new Uint8Array([1, 2]), "fid-1"),
  "empty tag": new FabricHash(new Uint8Array([1, 2]), ""),
}

--

tag fid1:
/Hash(tag:"fid1",hash:buf[0102])
/Hash(
  tag: "fid1",
  hash: buf [0102]
)
/Hash(
  tag: "fid1",
  hash: buf [0102]
)

tag outside the syntax:
/Hash(tag:"fid-1",hash:buf[0102])
/Hash(
  tag: "fid-1",
  hash: buf [0102]
)
/Hash(
  tag: "fid-1",
  hash: buf [0102]
)

empty tag:
/Hash(tag:"",hash:buf[0102])
/Hash(
  tag: "",
  hash: buf [0102]
)
/Hash(
  tag: "",
  hash: buf [0102]
)

(() => {
  class Data {
    x = 1;
    y = "two";
    nested = { z: [3] };
  }

  // A class name that is not a valid identifier renders quoted, as a key
  // does; a class with no name renders under the `<anonymous>` marker.
  class Renamed {
    static name = "Not a proper identifier";
    x = 1;
  }
  class WithNewline {
    static name = "a\nb";
  }
  return {
    data: new Data(),
    empty: new (class Empty {})(),
    "name not an identifier": new Renamed(),
    "name with a newline": new WithNewline(),
  };
})()
